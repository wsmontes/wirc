# Wirc v1.0 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Wirc v1.0 — a polished, testable iOS app that unifies RSS, YouTube, Podcasts, Mastodon, and IRC with onboarding, bookmarks, share extension, widget, and proper feed curation.

**Architecture:** 14 tasks across 4 phases. Each phase produces independently testable deliverables. Phase 1 restructures navigation and adds onboarding/bookmarks (core UX). Phase 2 extends the app with Share Extension and Widget targets. Phase 3 hardens Mastodon and search. Phase 4 adds tests and ships.

**Tech Stack:** Swift 6, SwiftUI, iOS 17+, `@Observable`, `NWConnection` (IRC), `URLSession` (HTTP/Mastodon), `XMLParser` (RSS), `WidgetKit`, `AppIntents`

## Global Constraints

- iOS 17.0 minimum deployment target
- Swift 6 concurrency safety — no new `@unchecked Sendable` without justification
- All UI must use `DesignSystem.Colors` and `DesignSystem.Fonts` tokens
- Build must succeed before each commit: `xcodebuild -project Wirc/Wirc.xcodeproj -scheme Wirc -destination 'platform=iOS Simulator,id=D3A8E60A-D820-4E29-A7E3-BC32DE7AD990' build`
- No feature removal without spec approval
- Follow existing patterns: `@Observable` + `@MainActor` for state, async/await for I/O, LazyVStack for lists
- Commit granularly — one commit per task

```swift
func toggleBookmark(_ object: WOMObject) {
    let existingId = womObjects.first(where: {
        $0.type.contains("wom:Signal") && $0.data["signalType"] == "bookmarked" && $0.data["targetId"] == object.id
    })?.id
    if let eid = existingId {
        // Remove bookmark
        womObjects.removeAll { $0.id == eid }
        Task { try? await store.delete(id: eid) }
    } else {
        // Add bookmark
        let signal = WOMObject(
            id: WOMIDGenerator.generate(type: "signal"),
            type: ["wom:Signal"],
            createdAt: Date(),
            data: ["signalType": "bookmarked", "targetId": object.id, "targetTitle": object.name ?? object.content?.text?.prefix(100) ?? ""],
            provenance: .localUser()
        )
        Task { try? await store.save(signal); await MainActor.run { womObjects.append(signal) } }
    }
}

func isBookmarked(_ object: WOMObject) -> Bool {
    womObjects.contains(where: {
        $0.type.contains("wom:Signal") && $0.data["signalType"] == "bookmarked" && $0.data["targetId"] == object.id
    })
}

import UIKit
import Social
import UniformTypeIdentifiers

class ShareViewController: UIViewController {
    override func viewDidLoad() {
        super.viewDidLoad()
        guard let item = extensionContext?.inputItems.first as? NSExtensionItem,
              let provider = item.attachments?.first else {
            complete()
            return
        }

        provider.loadItem(forTypeIdentifier: UTType.url.identifier) { [weak self] url, error in
            guard let self, let url = url as? URL else { self?.complete(); return }
            DispatchQueue.main.async {
                self.showConfirmation(for: url)
            }
        }
    }

    private func showConfirmation(for url: URL) {
        let label = UILabel()
        label.text = "Adding \(url.host ?? "feed")..."
        label.textAlignment = .center
        view.backgroundColor = .systemBackground
        view.addSubview(label)
        label.frame = view.bounds

        // Try to discover feed
        Task {
            let fetcher = FeedFetcher()
            if let discovered = try? await fetcher.discoverFeed(from: url) {
                let store = FeedSubscriptionStore()
                for feedURL in discovered {
                    let sub = FeedSubscription(feedURL: feedURL.absoluteString)
                    store.add(sub)
                }
                label.text = "Added \(discovered.count) feed(s)"
            } else {
                label.text = "No feed found at this URL"
            }
            try? await Task.sleep(for: .seconds(1.5))
            complete()
        }
    }

    private func complete() {
        extensionContext?.completeRequest(returningItems: [])
    }
}
```

import WidgetKit
import SwiftUI

struct Provider: TimelineProvider {
    func placeholder(in context: Context) -> SimpleEntry {
        SimpleEntry(cards: [
            CardData(title: "Example Article", source: "RSS", time: "2m ago"),
            CardData(title: "Latest Episode", source: "Podcast", time: "1h ago"),
            CardData(title: "New Release", source: "GitHub", time: "3h ago")
        ])
    }

    func getSnapshot(in context: Context, completion: @escaping (SimpleEntry) -> Void) {
        completion(placeholder(in: context))
    }

    func getTimeline(in context: Context, completion: @escaping (Timeline<SimpleEntry>) -> Void) {
        let store = JSONFileStore()
        Task {
            let objects = (try? await store.all()) ?? []
            let cards = objects
                .filter { $0.type.contains("wom:Post") }
                .sorted { $0.createdAt > $1.createdAt }
                .prefix(3)
                .map { CardData(title: $0.name ?? "", source: $0.data["network"] ?? "", time: $0.createdAt.formatted(.relative(presentation: .named))) }
            let entry = SimpleEntry(cards: cards)
            let nextUpdate = Date().addingTimeInterval(15 * 60)
            let timeline = Timeline(entries: [entry], policy: .after(nextUpdate))
            completion(timeline)
        }
    }
}

struct SimpleEntry: TimelineEntry {
    let date: Date = .now
    let cards: [CardData]

static func oauthURL(instance: String) throws -> URL {
    guard let base = URL(string: "https://\(instance)") else { throw MastodonError.invalidURL(instance) }
    var components = URLComponents(url: base.appendingPathComponent("/oauth/authorize"), resolvingAgainstBaseURL: false)
    components?.queryItems = [
        URLQueryItem(name: "client_id", value: "YOUR_CLIENT_ID"),  // Register a client or use dynamic registration
        URLQueryItem(name: "redirect_uri", value: "wirc://oauth/callback"),
        URLQueryItem(name: "scope", value: "read write"),
        URLQueryItem(name: "response_type", value: "code"),
    ]
    guard let url = components?.url else { throw MastodonError.invalidURL(instance) }
    return url
}
